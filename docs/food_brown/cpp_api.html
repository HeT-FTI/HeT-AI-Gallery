3. C++ API 指南

3.1. nnseg 核心类型

定义于 include/nnseg.hpp 与 include/nnseg_types.hpp。

  • nnseg::NNSegEngine

  • nnseg::SegConfig

  • nnseg::ImageFrameView

  • nnseg::SegmentationResult

#include "nnseg.hpp"
#include "nnseg_types.hpp"

nnseg::NNSegEngine engine;
nnseg::SegConfig cfg;
cfg.model_path = "model.onnx";
cfg.model_type = "yolov11";

std::string err;
if (!engine.Init(cfg, &err)) {
    // handle init error
}

nnseg::ImageFrameView frame;
auto seg = engine.Predict(frame);

3.2. food_browning 核心类型

定义于 include/food_browning.hpp。

  • food_browning::FoodBrowningEngine

  • food_browning::BrowningConfig

  • food_browning::BrowningResult

#include "food_browning.hpp"

food_browning::FoodBrowningEngine browning;
food_browning::BrowningConfig bcfg;

std::string err;
if (!browning.Init(bcfg, &err)) {
    // handle init error
}

auto result = browning.Analyze(frame, seg);
if (!result.ok) {
    // handle runtime error
}
