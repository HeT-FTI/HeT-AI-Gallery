4. C API 指南

4.1. nnseg C 接口

定义于 include/nnseg_c.h。

  • 生命周期：nnseg_create -> nnseg_init/nnseg_init_from_file -> nnseg_predict -> nnseg_result_release -> nnseg_destroy

  • 状态枚举：nnseg_status_t

nnseg_engine_t *engine = nnseg_create();
if (!engine) {
    return NNSEG_STATUS_NO_MEMORY;
}

nnseg_config_t cfg = {0};
cfg.model_path = "model.onnx";
cfg.model_type = "yolov11";

int32_t st = nnseg_init(engine, &cfg);
if (st != NNSEG_STATUS_OK) {
    nnseg_destroy(engine);
    return st;
}

4.2. food_browning C 接口

定义于 include/food_browning_c.h。

  • 生命周期：food_browning_create -> food_browning_init/food_browning_init_from_file -> food_browning_analyze -> food_browning_result_release -> food_browning_destroy

  • 状态枚举：food_browning_status_t

food_browning_engine_t *engine = food_browning_create();
if (!engine) {
    return FOOD_BROWNING_STATUS_NO_MEMORY;
}

food_browning_config_t cfg = {0};
int32_t st = food_browning_init(engine, &cfg);
if (st != FOOD_BROWNING_STATUS_OK) {
    food_browning_destroy(engine);
    return st;
}
